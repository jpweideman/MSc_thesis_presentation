\begin{frame}{Limits and next steps}
\label{frm:limitations}
\begin{itemize}
\setlength{\itemsep}{0pt}
\item It does not demonstrate improved decisions, and it does not study descision making at all!.
\setlength{\itemsep}{8mm}
\item \textbf{One sampling chain per run.}
\begin{itemize}
\item Test several chains, larger budgets, and other samplers.
\end{itemize}
\item \textbf{One dataset with repeated labels.}
\begin{itemize}
\item Evaluate further datasets with repeated labels.
\end{itemize}
\item \textbf{The uncertainty terms need validation.}
\begin{itemize}
\item Test known label variation and practical decision tasks.
\end{itemize}
\end{itemize}
